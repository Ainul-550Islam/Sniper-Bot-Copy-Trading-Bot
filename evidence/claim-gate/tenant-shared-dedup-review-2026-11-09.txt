    53	/// Claim `event` in the event dedup. The memory backend is process-local;
    54	/// Redis and Postgres marks are shared while available, but the facade falls
    55	/// back to process-local marks on backend errors and entries can expire or be
    56	/// evicted. This is not an unconditional cross-replica exactly-once guarantee.
    57	///
    58	/// Process-local dedup does not elect an executor across replicas. Durable
    59	/// dedup is an early, non-released event filter; ownership is a separate,
    60	/// lease-based execution gate. These facts do not by themselves establish
    61	/// the cause of a prior workspace-only test failure.
    62	pub async fn claim(state: &Shared, event: &LeaderTradeEvent) -> DedupOutcome {
    63	    let key = event.dedup_key();
    64	    let outcome = if state.mark_copy_event_seen(&key).await {
    65	        DedupOutcome::Fresh
    66	    } else {
    67	        DedupOutcome::Duplicate
    68	    };
    69	    metrics::count_dedup(outcome.as_str());
    70	    outcome
    71	}

    73	/// Require a shared first-arrival decision. Unlike [`claim`], this fails
    74	/// closed when the facade is missing, memory-only, or the shared backend
    75	/// cannot be reached. A failed or ambiguous write never authorizes execution.
    76	pub async fn claim_shared(
    77	    state: &Shared,
    78	    event: &LeaderTradeEvent,
    79	) -> bot_core::error::BotResult<DedupOutcome> {
    80	    let store = state.dedup().ok_or_else(|| {
    81	        bot_core::error::BotError::config("clustered copy execution requires shared event dedup")
    82	    })?;
    83	    let fresh = store.mark_shared("copy_event", &event.dedup_key()).await?;
    84	    let outcome = if fresh {
    85	        DedupOutcome::Fresh
    86	    } else {
    87	        DedupOutcome::Duplicate
    88	    };
    89	    metrics::count_dedup(outcome.as_str());
    90	    Ok(outcome)
